cmake_minimum_required(VERSION 3.22)
project(NewIME VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
include(CTest)
find_package(Threads REQUIRED)

option(NEWIME_WITH_LLAMA "Build CPU Zenz inference with pinned llama.cpp" ON)
option(NEWIME_WITH_FCITX5 "Build the fcitx5 input method" ON)
option(NEWIME_SANITIZE "Enable address and undefined behavior sanitizers for NewIME" OFF)
set(NEWIME_LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp" CACHE PATH "llama.cpp source")

add_library(newime_core STATIC src/text.cpp src/romaji.cpp src/dictionary.cpp src/converter.cpp src/config.cpp src/mozc.cpp)
target_include_directories(newime_core PUBLIC include)
target_link_libraries(newime_core PUBLIC Threads::Threads)
target_compile_options(newime_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(newime_core PRIVATE
    NEWIME_DATA_DIR="${CMAKE_INSTALL_FULL_DATADIR}/newime"
    NEWIME_SOURCE_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data"
    NEWIME_LIBEXEC_DIR="${CMAKE_INSTALL_FULL_LIBEXECDIR}/newime"
    NEWIME_DEFAULT_MODEL_PATH="${CMAKE_CURRENT_SOURCE_DIR}/models/zenz-v3.2-xsmall-Q5_K_M.gguf")
add_library(newime_zenz STATIC src/zenz.cpp)
target_include_directories(newime_zenz PUBLIC include)
target_link_libraries(newime_zenz PUBLIC newime_core Threads::Threads)
target_compile_options(newime_zenz PRIVATE -Wall -Wextra -Wpedantic)
if(NEWIME_WITH_LLAMA)
    if(NOT EXISTS "${NEWIME_LLAMA_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "Run ./scripts/bootstrap.sh first, or configure -DNEWIME_WITH_LLAMA=OFF")
    endif()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static inference libraries" FORCE)
    foreach(flag GGML_CUDA GGML_VULKAN GGML_METAL GGML_BLAS GGML_OPENMP GGML_NATIVE
                 LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER
                 LLAMA_BUILD_COMMON LLAMA_BUILD_APP LLAMA_OPENSSL LLAMA_CONNECT LLAMA_BUILD_MTMD)
        set(${flag} OFF CACHE BOOL "Disabled for CPU-only NewIME" FORCE)
    endforeach()
    set(GGML_CPU ON CACHE BOOL "CPU backend" FORCE)
    add_subdirectory("${NEWIME_LLAMA_DIR}" "third_party/llama.cpp" EXCLUDE_FROM_ALL)
    target_compile_definitions(newime_zenz PRIVATE NEWIME_WITH_LLAMA=1)
    target_link_libraries(newime_zenz PRIVATE llama ggml-cpu)
endif()
if(NEWIME_SANITIZE)
    foreach(target newime_core newime_zenz)
        target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endforeach()
endif()

add_executable(newime-cli src/main.cpp)
target_link_libraries(newime-cli PRIVATE newime_core newime_zenz)
install(TARGETS newime-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS third_party/mozc/meltype_mozc_helper DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/newime OPTIONAL)
install(FILES third_party/mozc/MOZC-LICENSE.txt third_party/mozc/MOZC-CREDITS.html third_party/mozc/provenance.json DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/newime/mozc OPTIONAL)
install(FILES third_party/mozc/meltype_mozc_helper.cc third_party/mozc/BUILD.fragment third_party/mozc/MOZC_COMMIT third_party/mozc/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/newime/mozc-source OPTIONAL)
install(PROGRAMS third_party/mozc/build-mozc-helper.sh DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/newime/mozc-source OPTIONAL)
install(FILES data/base.tsv data/english.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/newime)
install(FILES data/system.nimedic data/system.nimedic.sources.json DESTINATION ${CMAKE_INSTALL_DATADIR}/newime OPTIONAL)
install(DIRECTORY data/licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/newime OPTIONAL)
install(FILES models/zenz-v3.2-xsmall-Q5_K_M.gguf DESTINATION ${CMAKE_INSTALL_DATADIR}/newime/models OPTIONAL)
if(NEWIME_WITH_LLAMA)
    install(FILES "${NEWIME_LLAMA_DIR}/LICENSE" DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/newime RENAME llama.cpp-LICENSE)
    install(DIRECTORY "${NEWIME_LLAMA_DIR}/licenses/" DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/newime/llama-dependencies)
    install(FILES "${NEWIME_LLAMA_DIR}/vendor/hash/sha256/LICENSE" DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/newime RENAME sha256-LICENSE)
    install(FILES "${NEWIME_LLAMA_DIR}/vendor/hash/xxhash/LICENSE" DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/newime RENAME xxhash-LICENSE)
    install(FILES "${NEWIME_LLAMA_DIR}/vendor/hash/rotate-bits/LICENSE.md" DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/newime RENAME rotate-bits-LICENSE)
endif()
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/newime)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/newime)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/newime/docs)
install(FILES reports/performance.json reports/conversion.json reports/mixed-case.json reports/live-mozc.json DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/newime/reports OPTIONAL)
install(FILES tests/conversion_cases.tsv tests/mixed_case_cases.tsv DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/newime/tests)

if(NEWIME_WITH_FCITX5)
    find_package(Fcitx5Core 5.1 REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_get_variable(FCITX5_SYSTEM_LIBDIR Fcitx5Core libdir)
    file(WRITE "${CMAKE_BINARY_DIR}/fcitx5-addon-dir" "${FCITX5_SYSTEM_LIBDIR}/fcitx5\n")
    add_library(newime MODULE src/fcitx/newime.cpp src/fcitx/factory.cpp)
    set_target_properties(newime PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/addon")
    target_link_libraries(newime PRIVATE newime_core newime_zenz Fcitx5::Core)
    target_compile_options(newime PRIVATE -Wall -Wextra -Wpedantic)
    if(UNIX AND NOT APPLE)
        target_link_options(newime PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/fcitx/exports.map")
        set_property(TARGET newime APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/fcitx/exports.map")
    endif()
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test/addon" "${CMAKE_BINARY_DIR}/test/inputmethod")
    configure_file(data/newime-addon.conf.in test/addon/newime.conf @ONLY)
    configure_file(data/newime.conf.in test/inputmethod/newime.conf @ONLY)
    install(TARGETS newime LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/fcitx5)
    install(FILES "${CMAKE_BINARY_DIR}/test/addon/newime.conf" DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/addon)
    install(FILES "${CMAKE_BINARY_DIR}/test/inputmethod/newime.conf" DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod)
    install(FILES data/newime.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()

    target_link_libraries(mozc_test PRIVATE newime_core)

    if(NEWIME_WITH_LLAMA AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/models/zenz-v3.2-xsmall-Q5_K_M.gguf")

        set_tests_properties(zenz_real_cpu PROPERTIES TIMEOUT 30)
    endif()
    if(NEWIME_WITH_FCITX5 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/fcitx_test.cpp")

        get_filename_component(FCITX5_HEADERS_PARENT "${FCITX5_SYSTEM_INCLUDEDIR}" DIRECTORY)
        add_executable(fcitx_test tests/fcitx_test.cpp)
        target_link_libraries(fcitx_test PRIVATE newime_core Fcitx5::Core)
        target_include_directories(fcitx_test PRIVATE "${FCITX5_HEADERS_PARENT}/Module/fcitx-module/testfrontend")
        target_compile_definitions(fcitx_test PRIVATE
            NEWIME_BINARY_DIR="${CMAKE_BINARY_DIR}"
            NEWIME_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
            NEWIME_SYSTEM_ADDON_DIR="${FCITX5_SYSTEM_LIBDIR}/fcitx5")
        add_dependencies(fcitx_test newime)
        add_test(NAME fcitx_integration COMMAND fcitx_test)
        set_tests_properties(fcitx_integration PROPERTIES ENVIRONMENT "NEWIME_ZENZ=0;NEWIME_LEARNING=0" TIMEOUT 30)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mozc/meltype_mozc_helper")
            add_executable(fcitx_live_test tests/fcitx_live_test.cpp)
            target_link_libraries(fcitx_live_test PRIVATE newime_core Fcitx5::Core)
            target_include_directories(fcitx_live_test PRIVATE "${FCITX5_HEADERS_PARENT}/Module/fcitx-module/testfrontend")
            target_compile_definitions(fcitx_live_test PRIVATE
                NEWIME_BINARY_DIR="${CMAKE_BINARY_DIR}" NEWIME_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
                NEWIME_SYSTEM_ADDON_DIR="${FCITX5_SYSTEM_LIBDIR}/fcitx5")
            if(NEWIME_WITH_LLAMA AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/models/zenz-v3.2-xsmall-Q5_K_M.gguf")
                target_compile_definitions(fcitx_live_test PRIVATE NEWIME_TEST_NEURAL=1)
            endif()
            add_dependencies(fcitx_live_test newime)
            add_test(NAME fcitx_live_mozc COMMAND fcitx_live_test)
            set_tests_properties(fcitx_live_mozc PROPERTIES TIMEOUT 20)
        endif()
        if(Python3_Interpreter_FOUND)
            add_executable(launcher_test tests/launcher_test.cpp)
            target_link_libraries(launcher_test PRIVATE newime_core Fcitx5::Core)
            add_dependencies(launcher_test newime)
            add_test(NAME fcitx_launcher COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/launcher_test.py"
                "${CMAKE_BINARY_DIR}" "$<TARGET_FILE:launcher_test>")

        if(NEWIME_WITH_LLAMA AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/models/zenz-v3.2-xsmall-Q5_K_M.gguf")
            add_executable(fcitx_model_test tests/fcitx_model_test.cpp)
            target_link_libraries(fcitx_model_test PRIVATE newime_core Fcitx5::Core)
            target_include_directories(fcitx_model_test PRIVATE "${FCITX5_HEADERS_PARENT}/Module/fcitx-module/testfrontend")
