cmake_minimum_required(VERSION 3.22)
project(MonakIME VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)

find_package(Threads REQUIRED)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
# Preserve settings in pre-rename build directories and old setup commands.
foreach(name WITH_LLAMA WITH_FCITX5 SANITIZE CARGO_OFFLINE)
    if(DEFINED NEWIME_${name} AND NOT DEFINED MONAKIME_${name})
        set(MONAKIME_${name} "${NEWIME_${name}}" CACHE BOOL "Migrated NewIME option")
    endif()
endforeach()
if(DEFINED NEWIME_LLAMA_DIR AND NOT DEFINED MONAKIME_LLAMA_DIR)
    set(MONAKIME_LLAMA_DIR "${NEWIME_LLAMA_DIR}" CACHE PATH "Migrated llama.cpp source")
endif()
option(MONAKIME_WITH_LLAMA "Build CPU Zenz with pinned llama.cpp" ON)
option(MONAKIME_WITH_FCITX5 "Build the fcitx5 native adapter" ON)
option(MONAKIME_SANITIZE "Sanitize native adapters and fcitx integration tests" OFF)
option(MONAKIME_CARGO_OFFLINE "Use only cached Cargo dependencies" OFF)
set(MONAKIME_LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp" CACHE PATH "llama.cpp source")
if(MONAKIME_WITH_LLAMA)
    if(NOT EXISTS "${MONAKIME_LLAMA_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "Run ./scripts/bootstrap.sh first, or configure -DMONAKIME_WITH_LLAMA=OFF")
    endif()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static inference libraries" FORCE)
    foreach(flag GGML_CUDA GGML_VULKAN GGML_METAL GGML_BLAS GGML_OPENMP GGML_NATIVE
                 LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER
                 LLAMA_BUILD_COMMON LLAMA_BUILD_APP LLAMA_OPENSSL LLAMA_CONNECT LLAMA_BUILD_MTMD)
        set(${flag} OFF CACHE BOOL "Disabled for CPU-only MonakIME" FORCE)
    endforeach()
    set(GGML_CPU ON CACHE BOOL "CPU backend" FORCE)
    add_subdirectory("${MONAKIME_LLAMA_DIR}" "third_party/llama.cpp" EXCLUDE_FROM_ALL)
    add_library(monakime_llama STATIC native/llama_bridge.cpp)
    target_link_libraries(monakime_llama PRIVATE llama ggml-cpu)
endif()

set(RUST_TARGET_DIR "${CMAKE_BINARY_DIR}/cargo")
set(RUST_NATIVE_DIR "${CMAKE_BINARY_DIR}/rust-native")
file(MAKE_DIRECTORY "${RUST_NATIVE_DIR}")
set(RUST_FEATURES)
set(RUST_OFFLINE)
set(RUST_DEPENDS)
set(RUST_COPY_NATIVE)
if(MONAKIME_CARGO_OFFLINE)
    set(RUST_OFFLINE --offline)
endif()
if(MONAKIME_WITH_LLAMA)
    set(RUST_FEATURES --features llama)
    foreach(target monakime_llama llama ggml ggml-cpu ggml-base)
        list(APPEND RUST_DEPENDS ${target})
        list(APPEND RUST_COPY_NATIVE COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:${target}>" "${RUST_NATIVE_DIR}/lib${target}.a")
    endforeach()
endif()
set(RUST_ENV ${CMAKE_COMMAND} -E env
    "MONAKIME_NATIVE_DIR=${RUST_NATIVE_DIR}"
    "MONAKIME_INSTALL_DATA=${CMAKE_INSTALL_FULL_DATADIR}/monakime"
    "MONAKIME_INSTALL_LIBEXEC=${CMAKE_INSTALL_FULL_LIBEXECDIR}/monakime")
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.rs")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/monakime-cli" "${RUST_TARGET_DIR}/release/libmonakime.a"
    ${RUST_COPY_NATIVE}
    COMMAND ${RUST_ENV} ${CARGO_EXECUTABLE} build --release --locked ${RUST_OFFLINE}
        --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml" --target-dir "${RUST_TARGET_DIR}" ${RUST_FEATURES} --lib --bin monakime-cli
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RUST_TARGET_DIR}/release/monakime-cli" "${CMAKE_BINARY_DIR}/monakime-cli"
    COMMAND ${CMAKE_COMMAND} -E rm -f "${CMAKE_BINARY_DIR}/newime-cli"
    COMMAND ${CMAKE_COMMAND} -E create_symlink monakime-cli "${CMAKE_BINARY_DIR}/newime-cli"
    COMMAND ${CMAKE_COMMAND} -E touch "${CMAKE_BINARY_DIR}/monakime-cli" "${RUST_TARGET_DIR}/release/libmonakime.a"
    DEPENDS ${RUST_SOURCES} Cargo.toml Cargo.lock build.rs ${RUST_DEPENDS}
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" VERBATIM)
add_custom_target(monakime_rust_build ALL DEPENDS "${CMAKE_BINARY_DIR}/monakime-cli" "${RUST_TARGET_DIR}/release/libmonakime.a")
add_library(monakime_rust STATIC IMPORTED GLOBAL)
set_target_properties(monakime_rust PROPERTIES IMPORTED_LOCATION "${RUST_TARGET_DIR}/release/libmonakime.a")
add_dependencies(monakime_rust monakime_rust_build)
install(PROGRAMS "${CMAKE_BINARY_DIR}/monakime-cli" DESTINATION ${CMAKE_INSTALL_BINDIR})

if(MONAKIME_WITH_FCITX5)
    find_package(Fcitx5Core 5.1 REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_get_variable(FCITX5_SYSTEM_LIBDIR Fcitx5Core libdir)
    pkg_get_variable(FCITX5_SYSTEM_INCLUDEDIR Fcitx5Core includedir)
    get_filename_component(FCITX5_HEADERS_PARENT "${FCITX5_SYSTEM_INCLUDEDIR}" DIRECTORY)
    file(WRITE "${CMAKE_BINARY_DIR}/fcitx5-addon-dir" "${FCITX5_SYSTEM_LIBDIR}/fcitx5\n")
    add_library(monakime MODULE native/fcitx_bridge.cpp native/factory.cpp)
    set_target_properties(monakime PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/addon")
    target_link_libraries(monakime PRIVATE monakime_rust Fcitx5::Core Threads::Threads ${CMAKE_DL_LIBS} m)
    target_compile_options(monakime PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(monakime PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/native/exports.map")
    set_property(TARGET monakime APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native/exports.map")
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test/addon" "${CMAKE_BINARY_DIR}/test/inputmethod")
    configure_file(data/monakime-addon.conf.in test/addon/monakime.conf @ONLY)
    configure_file(data/monakime.conf.in test/inputmethod/monakime.conf @ONLY)
    configure_file(data/monakime-legacy.conf.in test/inputmethod/newime.conf @ONLY)
    file(REMOVE "${CMAKE_BINARY_DIR}/test/addon/newime.conf")
    install(TARGETS monakime LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/fcitx5)
    install(FILES "${CMAKE_BINARY_DIR}/test/addon/monakime.conf" DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/addon)
    install(FILES "${CMAKE_BINARY_DIR}/test/inputmethod/monakime.conf" DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod)
    install(FILES "${CMAKE_BINARY_DIR}/test/inputmethod/newime.conf" DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod)
    install(FILES data/monakime.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME rust COMMAND ${RUST_ENV} ${CARGO_EXECUTABLE} test --release --locked ${RUST_OFFLINE}
        --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml" --target-dir "${RUST_TARGET_DIR}" ${RUST_FEATURES} --all-targets)
    set_tests_properties(rust PROPERTIES TIMEOUT 120)
    add_test(NAME rust_real_backends COMMAND ${RUST_ENV} ${CARGO_EXECUTABLE} test --release --locked ${RUST_OFFLINE}
        --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml" --target-dir "${RUST_TARGET_DIR}" ${RUST_FEATURES} --test backends -- --ignored)
    set_tests_properties(rust_real_backends PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
    add_test(NAME dictionary_import COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/import_test.py" "${CMAKE_BINARY_DIR}/monakime-cli")
    add_test(NAME rename_compat COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/rename_test.py" "${CMAKE_BINARY_DIR}/monakime-cli")
    if(MONAKIME_WITH_FCITX5)
        set(FCITX_TESTS fcitx_test fcitx_live_test launcher_test)
        add_executable(shift_toggle_probe tests/shift_toggle_test.cpp)
        target_link_libraries(shift_toggle_probe PRIVATE Fcitx5::Core)
        add_test(NAME shift_toggle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/shift_toggle_test.py" "$<TARGET_FILE:shift_toggle_probe>")
        if(MONAKIME_WITH_LLAMA)
            list(APPEND FCITX_TESTS fcitx_model_test)
        endif()
        foreach(target IN LISTS FCITX_TESTS)
            add_executable(${target} "tests/${target}.cpp")
            target_link_libraries(${target} PRIVATE Fcitx5::Core)
            target_include_directories(${target} PRIVATE "${FCITX5_HEADERS_PARENT}/Module/fcitx-module/testfrontend")
            target_compile_definitions(${target} PRIVATE MONAKIME_BINARY_DIR="${CMAKE_BINARY_DIR}" MONAKIME_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" MONAKIME_SYSTEM_ADDON_DIR="${FCITX5_SYSTEM_LIBDIR}/fcitx5")
            add_dependencies(${target} monakime)
        endforeach()
        if(MONAKIME_WITH_LLAMA)
            target_compile_definitions(fcitx_live_test PRIVATE MONAKIME_TEST_NEURAL=1)
            add_test(NAME fcitx_async_cpu_model COMMAND fcitx_model_test)
            # These tests assert a real CPU inference deadline. Concurrent Cargo
            # compilation or another model test can consume that budget.
            set_tests_properties(fcitx_async_cpu_model PROPERTIES RUN_SERIAL TRUE TIMEOUT 30)
        endif()
        add_test(NAME fcitx_integration COMMAND fcitx_test)
        add_test(NAME fcitx_live_mozc COMMAND fcitx_live_test)
        add_test(NAME fcitx_launcher COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/launcher_test.py" "${CMAKE_BINARY_DIR}" "$<TARGET_FILE:launcher_test>")
        set_tests_properties(fcitx_integration fcitx_live_mozc fcitx_launcher PROPERTIES TIMEOUT 30)
        set_tests_properties(fcitx_live_mozc PROPERTIES RUN_SERIAL TRUE)
    endif()
endif()
if(MONAKIME_SANITIZE)
    foreach(target monakime monakime_llama ${FCITX_TESTS})

            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endforeach()
endif()
